function siteFromValue({ lineNumber, kind, unit, value }) {
  const specValue = unit === 's' ? spec.offlineAfterSeconds : spec.offlineAfterMilliseconds
  if (NUMERIC.test(value)) {
    const written = parseNumeric(value)
    return {
      lineNumber, kind, unit, cls: 'literal', written,
      agrees: written === specValue,
      note: written === specValue ? null : 'written value differs from spec',
    }

  const derived = value.match(DERIVED_CALL)
  if (derived !== null) {
    const unitArg = derived[1]
